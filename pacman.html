// جایگزین بخش مدیریت منطق روح‌ها در تابع update for(let g of ghosts) { // روح‌ها حالا در هر لحظه، جهت‌هایی که آن‌ها را به پک‌من نزدیک‌تر می‌کند اولویت می‌دهند let dirs = [{dx: g.speed, dy: 0}, {dx: -g.speed, dy: 0}, {dx: 0, dy: g.speed}, {dx: 0, dy: -g.speed}]; // فیلتر کردن مسیرهای مسدود let validDirs = dirs.filter(d => !isCollidingWithWall(g.x + d.dx, g.y + d.dy)); // هوش مصنوعی تهاجمی: همیشه بهترین جهت برای کاهش فاصله تا پک‌من را انتخاب کن if(validDirs.length > 0) { let bestDir = validDirs[0]; let minDist = Infinity; for (let d of validDirs) { let nextX = g.x + d.dx; let nextY = g.y + d.dy; let dist = Math.sqrt((nextX - pacman.x)**2 + (nextY - pacman.y)**2); // روح قرمز (قرمز) همیشه دنبالت می‌آید، بقیه روح‌ها کمی تصادفی‌تر اما تهاجمی let randomness = (g.color === "#ff0000") ? 0 : 0.3; if (dist < minDist && Math.random() > randomness) { minDist = dist; bestDir = d; } } g.dx = bestDir.dx; g.dy = bestDir.dy; } // اعمال حرکت g.x += g.dx; g.y += g.dy; // بررسی برخورد let dist = Math.sqrt((g.x - pacman.x)**2 + (g.y - pacman.y)**2); if(dist < 12) { gameOver = true; playSound('lose'); setTimeout(() => { location.reload(); }, 500); } }
